import { env as runtimeEnv, fetch as runtimeFetch } from "@executablemd/runtime";

function* run() {
  const repo = yield* runtimeEnv("GITHUB_REPOSITORY");
  if (!repo) {
    return "";
  }

  const api = `https://api.github.com/repos/${repo}`;
  function* request(path, options = {}) {
    const response = yield* runtimeFetch(`${api}${path}`, {
      ...options,
      headers: options.headers,
    });
    const body = yield* response.text();
    if (response.status < 200 || response.status >= 300) {
      throw new Error(`GitHub cleanup request failed (${response.status})`);
    }
    return body ? JSON.parse(body) : {};
  }

  const label = "cleanup";
  const topN = 5;
  const labelResponse = yield* runtimeFetch(`${api}/labels/${label}`);
  if (labelResponse.status === 404) {
    yield* request("/labels", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        name: label,
        description: "Auto-generated cleanup finding from repo analysis",
        color: "d4c5f9",
      }),
    });
  }

  const existingIssues = [];
  for (let page = 1; ; page++) {
    const batch = yield* request(`/issues?labels=${label}&state=open&per_page=100&page=${page}`);
    if (!Array.isArray(batch) || batch.length === 0) {
      break;
    }
    existingIssues.push(...batch);
    if (batch.length < 100) {
      break;
    }
  }

  const markerRe = /<!-- xmd-cleanup:(.+?) -->/;
  const issuesByFile = new Map();
  for (const issue of existingIssues) {
    const match = issue.body ? issue.body.match(markerRe) : null;
    if (match) {
      issuesByFile.set(match[1], issue);
    }
  }

  const topClusters = cleanupAnalysis.fileClusters.slice(0, topN);
  const topFiles = new Set(topClusters.map((cluster) => cluster.file));
  let created = 0;
  let updated = 0;
  let closed = 0;
  for (const cluster of topClusters) {
    const marker = `<!-- xmd-cleanup:${cluster.file} -->`;
    const title = `cleanup(${cluster.kind}): ${cluster.file} — ${cluster.totalViolations} violations, ${cluster.coOccurrence} rules`;
    const body = [
      marker,
      "",
      `## Cleanup: \`${cluster.file}\``,
      "",
      "| Metric | Value |",
      "|---|---|",
      `| Score | ${cluster.score} |`,
      `| Kind | ${cluster.kind} |`,
      `| Violations | ${cluster.totalViolations} |`,
      `| Co-occurring rules | ${cluster.coOccurrence} |`,
      `| Structural | ${cluster.categories.structural} |`,
      `| Verbosity | ${cluster.categories.verbosity} |`,
      `| Type-aware | ${cluster.categories.typeAware} |`,
      `| Other | ${cluster.categories.other} |`,
      "",
      `**Top rules:** ${cluster.ruleIds.slice(0, 15).join(", ")}`,
      "",
      "---",
      `<sub>Generated by executable.md repo analysis. Last updated: ${new Date().toISOString()}</sub>`,
    ].join("\n");
    const existing = issuesByFile.get(cluster.file);
    if (existing) {
      yield* request(`/issues/${existing.number}`, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ title, body }),
      });
      updated++;
    } else {
      yield* request("/issues", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ title, body, labels: [label] }),
      });
      created++;
    }
  }

  for (const [file, issue] of issuesByFile.entries()) {
    if (topFiles.has(file)) {
      continue;
    }
    yield* request(`/issues/${issue.number}/comments`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        body: "Resolved — file no longer in top-5 cleanup targets. Closing automatically.",
      }),
    });
    yield* request(`/issues/${issue.number}`, {
      method: "PATCH",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ state: "closed" }),
    });
    closed++;
  }
  return `Cleanup issues: created ${created}, updated ${updated}, closed ${closed}`;
}

return yield* run();
